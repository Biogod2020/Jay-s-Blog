\section{Selected Publications and Preprints}

\pub{1}{\textbf{Jia-Hao Ji}, Sijie Li, Jiabei Cheng, Zixi She, Jin-Tai Yu, and Zhiyuan Yuan. ``\textbf{Candidate supply and answer selection shape the value of LLM judging in multi-agent systems}.''}{arXiv:2608.25937 [cs.AI, cs.MA], 2026. \href{https://arxiv.org/abs/2608.25937}{Paper}}{First-author study separating candidate generation, answer recognition, and terminal selection.
